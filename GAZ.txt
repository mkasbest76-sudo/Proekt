PROGRAM GAZ
VAR
	imp_count     : UINT; (*Текущее значение из счетчика импульсов из модуля*)
	imp_last      : UINT:=1; (*Предыдущее значение из счетчика импульсов из модуля*)
    R1_imp_i: ARRAY[1..20] OF REAL; (*Массив мгновенных расходов в м3/час*)
	i: INT:=1; (*номер числа в массиве*)
    R1_imp: REAL; (*Средний мгновенный расход в м3/час*)
    LastTime : TIME:=T#32d13h53m5s20ms;       (* Время предыдущего импульса *)
    CurrTime : TIME;    (* Текущее системное время *)
    tDelta   : TIME;          (* Результат: время между импульсами *)
    diMillis : ARRAY[1..3] OF DINT;        (* Время в мс (числом) *)
	IF_1, IF_2: BOOL; (*Условие проверки достоверности времени между импульсами*)

	
T: REAL;	
	
END_VAR

	CurrTime := TIME();
	IF TIME_TO_DINT(CurrTime-LastTime)>60000 THEN  imp_last:= imp_count; LastTime := CurrTime;
		FOR i:=1 TO 19 DO R1_imp_i[21-i]:=0; R1_imp_i[1]:= 0;END_FOR
		R1_imp:=0; 
	ELSE IF imp_count<> imp_last AND LastTime <> T#0S THEN 
		FOR i:=1 TO 2 DO diMillis [4-i]:=diMillis [3-i];END_FOR
		FOR i:=1 TO 19 DO R1_imp_i[21-i]:=R1_imp_i[20-i]; END_FOR
      	tDelta:= CurrTime - LastTime; diMillis [1]:= TIME_TO_DINT(tDelta);
		IF DINT_TO_REAL(diMillis[1])>0 THEN 
			IF ABS(1-DINT_TO_REAL(diMillis[2])/DINT_TO_REAL(diMillis[1]))<0.2 THEN IF_1:=TRUE; ELSE IF_1:=FALSE; END_IF
			IF ABS(1-DINT_TO_REAL(diMillis[3])/DINT_TO_REAL(diMillis[1]))<0.2 THEN IF_2:=TRUE; ELSE IF_2:=FALSE; END_IF END_IF
			IF IF_1 AND IF_2 AND DINT_TO_REAL (diMillis[1])>0 AND (imp_count-imp_last)>0 THEN 
			R1_imp_i[1]:= 3600000/DINT_TO_REAL (diMillis[1])/(imp_count-imp_last); 
			ELSE R1_imp_i[1]:= R1_imp; END_IF
		R1_imp:= (R1_imp_i[1]+R1_imp_i[2]+R1_imp_i[3]+R1_imp_i[4]+R1_imp_i[5]+R1_imp_i[6]+R1_imp_i[7]+R1_imp_i[8]+R1_imp_i[9]+R1_imp_i[10]+R1_imp_i[11]+R1_imp_i[12]+R1_imp_i[13]+R1_imp_i[14]+R1_imp_i[15]+R1_imp_i[16]+R1_imp_i[17]+R1_imp_i[18]+R1_imp_i[19]+R1_imp_i[20])/20; 
    imp_last:= imp_count;
    LastTime := CurrTime;
	END_IF 
END_IF
